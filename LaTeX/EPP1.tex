\subsection{Basic quantities}
\begin{align}
	V &= \frac{W}{Q}, & I &= \frac{\mathrm{d}Q}{\mathrm{d}t}, & P &= VI.
\end{align}
Ohm's law is
\begin{equation}
	V = IR.
\end{equation}

\section{Thevenin Equivalent Circuits}

Any linear two-terminal network can be replaced by a Thevenin voltage
source $V_{\mathrm{th}}$ in series with a resistance $R_{\mathrm{th}}$:
\begin{equation}
	V_{\mathrm{th}} = V_{\mathrm{oc}},\qquad
	R_{\mathrm{th}} = \frac{V_{\mathrm{oc}}}{I_{\mathrm{sc}}}.
\end{equation}

\section{Node Voltage Analysis}

Choose a reference node, assign a voltage to every remaining node, and
apply Kirchhoff's current law:
\begin{equation}
	\sum_k \frac{V_n - V_k}{R_k} = I_{\mathrm{injected}}.
\end{equation}

\section{Capacitors}

For a capacitor,
\begin{equation}
	q = Cv,\qquad i = C\frac{\mathrm{d}v}{\mathrm{d}t},\qquad
	w = \frac{1}{2}Cv^2.
\end{equation}
For series and parallel combinations,
\begin{align}
	\frac{1}{C_{\mathrm{series}}} &= \sum_i \frac{1}{C_i}, &
	C_{\mathrm{parallel}} &= \sum_i C_i.
\end{align}

\section{Capacitors and Inductors}

An inductor satisfies
\begin{equation}
	v = L\frac{\mathrm{d}i}{\mathrm{d}t},\qquad
	w = \frac{1}{2}Li^2.
\end{equation}
The impedance of ideal reactive components at angular frequency $\omega$ is
\begin{equation}
	Z_C = \frac{1}{j\omega C},\qquad Z_L = j\omega L.
\end{equation}

\section{AC Circuits and Phasors}

For sinusoidal steady-state analysis, represent signals using phasors:
\begin{equation}
	v(t) = V_{\mathrm{m}}\cos(\omega t + \phi)
	\quad\longleftrightarrow\quad
	\tilde V = V_{\mathrm{rms}}\angle\phi.
\end{equation}
Impedances can be combined using the same series and parallel rules as
resistances. The average power is
\begin{equation}
	P = V_{\mathrm{rms}}I_{\mathrm{rms}}\cos\phi.
\end{equation}
